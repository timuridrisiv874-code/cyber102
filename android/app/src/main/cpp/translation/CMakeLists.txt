cmake_minimum_required(VERSION 3.24)
project(cyber_translation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_CLI OFF CACHE BOOL "" FORCE)
set(WITH_MKL OFF CACHE BOOL "" FORCE)
set(WITH_RUY ON CACHE BOOL "" FORCE)
set(OPENMP_RUNTIME NONE CACHE STRING "" FORCE)
set(RUY_MINIMAL_BUILD ON CACHE BOOL "" FORCE)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_NFKC_COMPILE OFF CACHE BOOL "" FORCE)
set(ABSL_PROPAGATE_CXX_STD ON CACHE BOOL "" FORCE)

add_subdirectory("${TRANSLATION_SOURCE}/ctranslate2" ct2 EXCLUDE_FROM_ALL)
set_source_files_properties("${TRANSLATION_SOURCE}/ctranslate2/src/thread_pool.cc"
    TARGET_DIRECTORY ctranslate2 PROPERTIES COMPILE_OPTIONS "-include;${CMAKE_CURRENT_SOURCE_DIR}/android_thread_affinity.h")
add_subdirectory("${TRANSLATION_SOURCE}/sentencepiece" spm EXCLUDE_FROM_ALL)
add_library(cyber_translation SHARED translation_jni.cpp)
target_include_directories(cyber_translation PRIVATE "${TRANSLATION_SOURCE}/sentencepiece/src")
target_link_libraries(cyber_translation PRIVATE ctranslate2 sentencepiece-static android log)
target_link_options(cyber_translation PRIVATE -Wl,--exclude-libs,ALL -Wl,-z,max-page-size=16384)
