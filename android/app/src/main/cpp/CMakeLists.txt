cmake_minimum_required(VERSION 3.22)
project(cyber_smol LANGUAGES C CXX)

# A debuggable Java APK still needs optimized ggml kernels. Keeping -O0 in
# dependencies makes on-device visual inference unusably slow.
set(CMAKE_BUILD_TYPE Release CACHE STRING "Optimized native inference" FORCE)
set_property(GLOBAL PROPERTY JOB_POOLS cyber_compile=2 cyber_link=1)
set(CMAKE_JOB_POOL_COMPILE cyber_compile)
set(CMAKE_JOB_POOL_LINK cyber_link)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CYBER_LLAMA_REVISION "b1b770d90d94cad9e9ea5b993bdbc383a522fb28")
get_filename_component(CYBER_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(LLAMA_CPP_DIR "${CYBER_ROOT}/var/native/llama.cpp-${CYBER_LLAMA_REVISION}")
if(NOT EXISTS "${LLAMA_CPP_DIR}/CYBER102_REVISION")
    message(FATAL_ERROR "Prepare pinned source first: python3 ${CMAKE_CURRENT_LIST_DIR}/fetch_llama.py")
endif()
file(READ "${LLAMA_CPP_DIR}/CYBER102_REVISION" CYBER_ACTUAL_REVISION)
string(STRIP "${CYBER_ACTUAL_REVISION}" CYBER_ACTUAL_REVISION)
if(NOT CYBER_ACTUAL_REVISION STREQUAL CYBER_LLAMA_REVISION)
    message(FATAL_ERROR "llama.cpp revision does not match this JNI adapter")
endif()

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_SUBPROCESS OFF CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
if(ANDROID AND ANDROID_ABI STREQUAL "arm64-v8a")
    # arm64-v8a includes phones without the optional dot-product instructions.
    set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
endif()
add_subdirectory("${LLAMA_CPP_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/llama" EXCLUDE_FROM_ALL)

add_library(cyber_smol_core STATIC smol_engine.cpp)
target_include_directories(cyber_smol_core PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(cyber_smol_core PUBLIC llama mtmd)
target_compile_options(cyber_smol_core PRIVATE -O3)

if(ANDROID)
    add_library(cyber_smol SHARED smol_jni.cpp)
    target_link_libraries(cyber_smol PRIVATE cyber_smol_core android log)
    target_compile_options(cyber_smol PRIVATE -O3)
    target_link_options(cyber_smol PRIVATE "-Wl,-z,max-page-size=16384")
else()
    add_executable(cyber-smol-cli "${CYBER_ROOT}/tests/native/smol_cli.cpp")
    target_link_libraries(cyber-smol-cli PRIVATE cyber_smol_core)
    add_executable(cyber-smol-smoke "${CYBER_ROOT}/tests/native/smol_smoke.cpp")
    target_link_libraries(cyber-smol-smoke PRIVATE cyber_smol_core)
    add_executable(cyber-smol-benchmark "${CYBER_ROOT}/tests/native/smol_benchmark.cpp")
    target_link_libraries(cyber-smol-benchmark PRIVATE cyber_smol_core)
endif()
